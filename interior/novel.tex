% bookpress — pandoc LaTeX template for a hand-bound novel.
%
% Page geometry, margins and gutter arrive as -V variables from build.sh,
% which reads them out of book.toml. Nothing here should be edited to change
% a measurement; edit book.toml instead.
%
% Engine: LuaLaTeX (fontspec + microtype protrusion both want it).
% Class:  memoir, for its trimmed-stock model and page-style machinery.

\documentclass[$if(fontsize)$$fontsize$$else$10.5pt$endif$,twoside,openright]{memoir}

% ---------------------------------------------------------------- geometry --
% The stock IS the trimmed page: we impose onto larger sheets later, so as
% far as LaTeX is concerned the paper is exactly the finished book page.
\setstocksize{$trimheight$}{$trimwidth$}
\settrimmedsize{\stockheight}{\stockwidth}{*}
\settrims{0pt}{0pt}

% Spine-side margin carries the gutter. On a hand-glued perfect binding the
% book will not open flat, so the inner margin you measure is not the inner
% margin you read — the gutter buys that back.
\setlrmarginsandblock{$innermargin$}{$outermargin$}{*}
\setulmarginsandblock{$topmargin$}{$bottommargin$}{*}
\setheadfoot{\onelineskip}{2\onelineskip}
\setheaderspaces{*}{1.2\onelineskip}{*}
\checkandfixthelayout

% ------------------------------------------------------------------- fonts --
\usepackage{fontspec}
\usepackage{microtype}

% Looked up through kpathsea by filename so this works whether or not
% fontconfig in the container knows about the texmf tree.
\setmainfont{EBGaramond}[
  Extension      = .otf,
  UprightFont    = *-Regular,
  ItalicFont     = *-Italic,
  BoldFont       = *-SemiBold,
  BoldItalicFont = *-SemiBoldItalic,
  Numbers        = OldStyle,
  Ligatures      = TeX,
]

\linespread{$linespread$}

% ------------------------------------------------------------- typesetting --
% A novel is a single column of text with nothing to distract the eye, so the
% only real work is keeping the rag and the page bottoms honest.
\midsloppy
\frenchspacing            % uniform inter-sentence space, as book printers set
\clubpenalty=10000        % no orphan: never one line alone at a page foot
\widowpenalty=10000       % no widow:  never one line alone at a page head
\displaywidowpenalty=10000
\brokenpenalty=10000      % never break a word across a page turn
\raggedbottom             % prefer a short page over stretched leading

\setlength{\parindent}{1.2em}
\nonzeroparskip\setlength{\parskip}{0pt}

\usepackage[hidelinks,pdfusetitle]{hyperref}

% ------------------------------------------------------------- page styles --
% Verso: page number outside, book title centred. Recto: chapter title
% centred, page number outside. Nothing in the feet.
\makepagestyle{bookpress}
\makeevenhead{bookpress}{\small\thepage}{\small\scshape $booktitle$}{}
\makeoddhead{bookpress}{}{\small\scshape\rightmark}{\small\thepage}
\makeevenfoot{bookpress}{}{}{}
\makeoddfoot{bookpress}{}{}{}
\makepsmarks{bookpress}{%
  \nouppercaseheads
  \createmark{chapter}{right}{shownumber}{}{. \space}
}
\pagestyle{bookpress}

% Chapter openings: a plain page, no running head, number spelled out above a
% small-caps title. Deliberately quiet — the drama belongs to the cover.
\makechapterstyle{bookpress}{%
  \setlength{\beforechapskip}{0pt}
  \setlength{\midchapskip}{1.5\onelineskip}
  \setlength{\afterchapskip}{2.5\onelineskip}
  \renewcommand{\chapnamefont}{\normalfont\small\scshape}
  \renewcommand{\chapnumfont}{\normalfont\small\scshape}
  \renewcommand{\chaptitlefont}{\normalfont\Large\scshape}
  \renewcommand{\printchaptername}{\centering\chapnamefont\chaptername}
  \renewcommand{\chapternamenum}{\space}
  \renewcommand{\printchapternum}{\chapnumfont\thechapter}
  \renewcommand{\afterchapternum}{\par\nobreak\vskip\midchapskip}
  \renewcommand{\printchaptertitle}[1]{\centering\chaptitlefont ##1}
}
\chapterstyle{bookpress}
\aliaspagestyle{chapter}{empty}

% ------------------------------------------------------- pandoc scaffolding --
% Definitions pandoc's LaTeX writer assumes exist.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\usepackage{graphicx}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>\textheight\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}

% Scene breaks. A centred ornament rather than a printer's rule, and the
% paragraph that follows it opens flush left — the ornament has already done
% the work of signalling the break, so an indent on top of it reads as noise.
% \@afterheading is what actually suppresses that indent; \noindent alone
% cannot, because pandoc emits the following text as its own paragraph.
\makeatletter
\newcommand{\scenebreak}{%
  \par\vspace{1.2\onelineskip}%
  \centerline{\normalfont\large *\quad*\quad*}%
  \par\vspace{1.2\onelineskip}%
  \@afterindentfalse\@afterheading}
\makeatother

$for(header-includes)$
$header-includes$
$endfor$

% The block must come out to an even number of pages: a leaf has two sides,
% and an odd count means a stray blank you did not choose. cleardoublepage
% at the end guarantees it.
\AtEndDocument{\cleardoublepage}

\begin{document}

% ---------------------------------------------------------------- frontmatter
\frontmatter
\pagestyle{empty}

% Half-title.
\begin{vplace}[0.35]
\begin{center}
  {\Large\scshape $booktitle$}
\end{center}
\end{vplace}
\cleardoublepage

% Title page.
\begin{vplace}[0.35]
\begin{center}
  {\HUGE\scshape $booktitle$}\par
  $if(subtitle)$\vspace{1.5\onelineskip}{\large\itshape $subtitle$}\par$endif$
  \vspace{3\onelineskip}
  {\large $author$}\par
  \vfill
  {\small\scshape $if(imprint)$$imprint$$else$Printed and bound by hand$endif$}\par
  $if(year)${\small $year$}\par$endif$
\end{center}
\end{vplace}
\cleardoublepage

% Colophon / public-domain note.
\thispagestyle{empty}
\begin{vplace}[0.75]
\begin{center}\small
  $booktitle$ was first published in $if(year)$$year$$else$the nineteenth century$endif$\\
  and is in the public domain.\par
  \vspace{\onelineskip}
  Set in EB Garamond. Printed on a desktop laser,\\
  imposed, folded, glued and trimmed by hand.\par
\end{center}
\end{vplace}
\cleardoublepage

$if(toc)$
\pagestyle{bookpress}
\tableofcontents*
\cleardoublepage
$endif$

% ----------------------------------------------------------------- mainmatter
\mainmatter
\pagestyle{bookpress}

$body$

\end{document}
